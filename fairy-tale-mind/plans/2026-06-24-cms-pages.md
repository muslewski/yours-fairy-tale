# CMS Pages Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `Pages` Payload collection composed of layout blocks, with per-page SEO and on-demand revalidation, resolved at `/[slug]` by the App Router — proven on one fresh page. Homepage and configurator untouched.

**Architecture:** New `pages` collection (drafts enabled) whose `layout` is a blocks field (Hero/RichText/Media/CTA). A `RenderBlocks` server component maps blockType→component. `app/(site)/[slug]/page.tsx` fetches a published page via the Local API and renders it; metadata comes from `@payloadcms/plugin-seo`. Caching + revalidation mirror `lib/pricing-source.ts` (`unstable_cache` + tag busting in collection hooks). Migration auto-applies on deploy via `instrumentation.ts`.

**Tech Stack:** Next.js 16.2.6 (App Router), React 19, Payload 3.85, `@payloadcms/plugin-seo` ^3.85, `@payloadcms/richtext-lexical/react`, Postgres (Neon), Tailwind v4 brand tokens.

## Global Constraints

- **Spec:** `fairy-tale-mind/specs/2026-06-24-cms-pages-design.md`.
- Colors: brand tokens only (`bg-brand-*`, `text-brand-*`, `border-brand-deep`); never hardcode hex. Comic shadow via `shadow-comic*`. Fonts: `var(--font-fredoka)` headings, `var(--font-fraunces)` editorial, `var(--font-quicksand)` body.
- Brand voice on any copy/labels: calm, parent-facing, sentence case, American English, no em-dash, no SFX, rare exclamation.
- Every CTA/button leads somewhere real — render no button when its `url` is empty (homepage invariant).
- Server components by default; a `"use client"` leaf only if a block needs interactivity (none of the four starter blocks do).

- Revalidation hooks must `await import("next/cache")` (dynamic) so `next/cache` stays out of the Payload CLI graph; Next 16 `revalidateTag(tag, "max")` two-arg form.
- Pin `@payloadcms/plugin-seo@^3.85.0`.
- SEO base URL: `https://www.yoursfairytale.com` (matches `app/sitemap.ts` + `(site)/layout.tsx` `metadataBase`).
- **Read before coding (AGENTS.md):** the relevant guides under `node_modules/next/dist/docs/` for `generateStaticParams`, `generateMetadata`, `draftMode`, and dynamic routes — Next 16 differs from training data (`draftMode()` is async).

// lib/reserved-slugs.ts
/**
 * Slugs a CMS Page may NOT claim — they map to real static routes (or the
 * homepage), and a Page must never shadow them. Used by the Pages `slug`
 * validate and by generateStaticParams.
 */
const CONCEPT_SLUGS = [
  "1-magic-sparkle", "2-bento-grid", "3-glass-dream", "4-storybook-editorial",
  "5-aurora-mesh", "6-pop-comic", "7-cloud-castle", "8-neumorph-pastel",
  "9-sticker-sheet", "10-floating-3d",
];

export const RESERVED_SLUGS: ReadonlySet<string> = new Set([
  "", "home",
  "blog", "contact", "series", "studio", "app", "admin", "api",
  "sign-in", "open", "order-confirmed", "legacy-examples",
  "privacy", "terms", "refund",
  ...CONCEPT_SLUGS,
]);

/** Lowercase, strip accents, collapse non-alphanumerics to single hyphens. */
export function normalizeSlug(input: string): string {
  return input
    .normalize("NFKD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

export function isReservedSlug(slug: string): boolean {
  return RESERVED_SLUGS.has(normalizeSlug(slug));
}
```

// collections/Pages/hooks/revalidate.ts
import type { CollectionAfterChangeHook, CollectionAfterDeleteHook } from "payload";

async function bust(slug: string) {
  // Dynamic import keeps next/cache out of the Payload CLI graph (migrate /
  // generate:types load this config outside a request). Mirrors globals/Pricing.
  const { revalidatePath, revalidateTag } = await import("next/cache");
  revalidatePath("/" + slug);
  revalidateTag("page:" + slug, "max");
  revalidateTag("pages-sitemap", "max");
}

export const revalidatePage: CollectionAfterChangeHook = async ({ doc, previousDoc, context }) => {
  if (context?.disableRevalidate) return doc;
  if (doc?.slug) await bust(doc.slug);
  // Slug renamed → also clear the old path/tag.
  if (previousDoc?.slug && previousDoc.slug !== doc?.slug) await bust(previousDoc.slug);
  return doc;
};

export const revalidatePageDelete: CollectionAfterDeleteHook = async ({ doc, context }) => {
  if (context?.disableRevalidate) return doc;
  if (doc?.slug) await bust(doc.slug);
  return doc;
};
```

import { Hero } from "@/blocks/Hero/config";

export const Pages: CollectionConfig = {
  slug: "pages",
  labels: { singular: "Page", plural: "Pages" },
  access: {
    // Public reads see published only; an authenticated admin sees drafts too.
    read: ({ req }) => (req.user ? true : { _status: { equals: "published" } }),
    create: adminOnly,
    update: adminOnly,
    delete: adminOnly,
  },
  admin: {
    group: "Content",
    useAsTitle: "title",
    defaultColumns: ["title", "slug", "_status", "updatedAt"],
  },
  versions: { drafts: { autosave: false } },
  hooks: {
    afterChange: [revalidatePage],
    afterDelete: [revalidatePageDelete],
  },
  fields: [
    { name: "title", type: "text", required: true },
    {
      name: "slug",
      type: "text",
      required: true,
      unique: true,
      index: true,
      admin: { position: "sidebar", description: 'URL after the domain, e.g. "about".' },
      hooks: { beforeValidate: [({ value }) => (typeof value === "string" ? normalizeSlug(value) : value)] },

      name: "layout",
      type: "blocks",
      minRows: 0,

// lib/pages-source.ts
import { unstable_cache } from "next/cache";

import { getPayloadClient } from "@/lib/payload";
import type { Page } from "@/payload-types";

export type PageDoc = Page;

async function readPage(slug: string, draft: boolean): Promise<PageDoc | null> {
  try {
    const payload = await getPayloadClient();
    const res = await payload.find({
      collection: "pages",
      draft,
      limit: 1,
      overrideAccess: draft, // drafts need access override; public reads stay gated
      where: draft
        ? { slug: { equals: slug } }
        : { and: [{ slug: { equals: slug } }, { _status: { equals: "published" } }] },
    });
    return (res.docs[0] as PageDoc) ?? null;
  } catch {
    return null;
  }
}

/** Cached published read; tagged so a publish (page:<slug>) busts it. Draft
 *  reads bypass the cache (preview is request-time, never cached). */
export async function getPageBySlug(slug: string, opts?: { draft?: boolean }): Promise<PageDoc | null> {
  if (opts?.draft) return readPage(slug, true);
  const cached = unstable_cache(() => readPage(slug, false), ["page", slug], {
    tags: ["pages", "page:" + slug],
    revalidate: 300,
  });
  return cached();
}

async function readPublishedSlugs(): Promise<string[]> {
  try {
    const payload = await getPayloadClient();
    const res = await payload.find({
      collection: "pages",
      where: { _status: { equals: "published" } },
      limit: 1000,
      pagination: false,
      select: { slug: true },
    });
    return res.docs.map((d) => (d as { slug: string }).slug).filter(Boolean);
  } catch {
    return [];
  }
}

export const getPublishedPageSlugs = unstable_cache(readPublishedSlugs, ["pages-slugs"], {
  tags: ["pages", "pages-sitemap"],
  revalidate: 300,
});
```

import type { Page } from "@/payload-types";
import { HeroRender } from "./hero";

type Block = NonNullable<Page["layout"]>[number];

export function RenderBlocks({ blocks }: { blocks?: Page["layout"] }) {
  if (!Array.isArray(blocks) || blocks.length === 0) return null;
  return (
    <>
      {blocks.map((block: Block, i: number) => {
        const key = ("id" in block && block.id) || i;

          case "richText":
            return <RichTextRender key={key} block={block} />;
          case "mediaBlock":
            return <MediaRender key={key} block={block} />;
          case "cta":
            return <CTARender key={key} block={block} />;
          default:
            return null; // forward-compat: unknown block types render nothing
        }
      })}
    </>
  );

// app/(site)/[slug]/page.tsx
import type { Metadata } from "next";
import { draftMode } from "next/headers";
import { notFound } from "next/navigation";

import { SiteNav } from "@/components/home/site-nav";
import { SiteFooter } from "@/components/home/site-footer";
import { RenderBlocks } from "@/components/blocks/render-blocks";
import { getPageBySlug, getPublishedPageSlugs } from "@/lib/pages-source";

export async function generateStaticParams() {
  const slugs = await getPublishedPageSlugs();
  return slugs.map((slug) => ({ slug }));
}

type Params = { params: Promise<{ slug: string }> };

export async function generateMetadata({ params }: Params): Promise<Metadata> {
  const { slug } = await params;
  const { isEnabled: draft } = await draftMode();
  const page = await getPageBySlug(slug, { draft });
  if (!page) return {};
  const meta = (page as { meta?: { title?: string; description?: string; image?: { url?: string } | string } }).meta;
  const title = meta?.title ?? page.title;
  const description = meta?.description ?? undefined;
  const ogUrl = typeof meta?.image === "object" ? meta?.image?.url : undefined;
  return {
    title,
    description,
    alternates: { canonical: "/" + slug },
    openGraph: { title, description, url: "/" + slug, ...(ogUrl ? { images: [{ url: ogUrl }] } : {}) },
  };
}

export default async function CmsPage({ params }: Params) {
  const { slug } = await params;
  const { isEnabled: draft } = await draftMode();
  const page = await getPageBySlug(slug, { draft });
  if (!page) notFound();
  return (
    <>
      <SiteNav />
      <main className="font-[family-name:var(--font-quicksand)] text-brand-deep">
        <RenderBlocks blocks={page.layout} />
      </main>
      <SiteFooter />
    </>
  );
}
```

A per-slug dynamic OG via `lib/og.tsx` (an `opengraph-image.tsx` in this dir) is optional polish — add only if `lib/og.tsx` exposes a reusable generator; otherwise the layout default suffices for sub-project 1. Confirm `SiteNav`/`SiteFooter` import paths match the homepage.
